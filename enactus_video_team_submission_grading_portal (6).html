function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var props = PropertiesService.getScriptProperties();
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var body = JSON.parse(e.postData.contents);

    // 1. Save Full Website State (Members, Tasks, Drive Links, Grades, Matrix)
    if (body.fullPortalState) {
      props.setProperty("PORTAL_STATE_JSON", JSON.stringify(body.fullPortalState));
    }

    // 2. Also Log/Update Row in Google Sheet for Easy Viewing
    if (body.record) {
      var data = body.record;
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Timestamp", "TaskID", "MemberID", "MemberCode", "MemberName", "DriveLink", "Note", "Score", "Evaluator", "Feedback"]);
      }
      var rows = sheet.getDataRange().getValues();
      var updated = false;
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][1]) === String(data.taskId) && (String(rows[i][3]) === String(data.memberCode) || String(rows[i][2]) === String(data.memberId))) {
          sheet.getRange(i + 1, 1, 1, 10).setValues([[
            data.timestamp || new Date().toISOString().slice(0,16).replace("T"," "),
            data.taskId || "",
            data.memberId || rows[i][2] || "",
            data.memberCode || rows[i][3] || "",
            data.memberName || rows[i][4] || "",
            data.driveLink !== undefined ? data.driveLink : rows[i][5],
            data.note !== undefined ? data.note : rows[i][6],
            data.score !== undefined && data.score !== null ? data.score : rows[i][7],
            data.evaluator !== undefined ? data.evaluator : rows[i][8],
            data.feedback !== undefined ? data.feedback : rows[i][9]
          ]]);
          updated = true;
          break;
        }
      }
      if (!updated) {
        sheet.appendRow([
          data.timestamp || new Date().toISOString().slice(0,16).replace("T"," "),
          data.taskId || "",
          data.memberId || "",
          data.memberCode || "",
          data.memberName || "",
          data.driveLink || "",
          data.note || "",
          data.score !== undefined && data.score !== null ? data.score : "",
          data.evaluator || "",
          data.feedback || ""
        ]);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "OK" })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var props = PropertiesService.getScriptProperties();
  var savedState = props.getProperty("PORTAL_STATE_JSON");
  return ContentService.createTextOutput(savedState ? savedState : JSON.stringify({ empty: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
